Thông tin sơ bộ về HW-504:

Thông tin cơ bản:

\(\bullet\) Đây là 1 con nút tay cầm chơi game

\(\bullet\) Điện áp hoạt động: 5V

\(\bullet\) Không cần gắn thêm điện trở

\(\bullet\) Sơ đồ nối chân cơ bản(giữa chân cảm biến-arduino)

- Cực dương(+5V) nối với đầu 5V

- Cực âm(GND) nối với đầu GND

- Chân VRx nối với chân A0(hoặc bất kì chân analog nào của arduino)

- Chân VRy nối với chân A1(hoặc bất kì chân analog nào của arduino)

- Chân SW nối với chân 2(hoặc bất kì chân digital nào của arduino)

- Code mẫu:

// Định nghĩa các chân kết nối cảm biến Joystick HW-504

const int pinX = A0; // Chân VRx nối với Analog A0

const int pinY = A1; // Chân VRy nối với Analog A1

const int pinSW = 2; // Chân SW nối với Digital 2 (Hỗ trợ Ngắt hệ thống)

void setup() {

__Serial.begin(9600); // Mở cổng Serial Monitor xem tọa độ di chuyển

__// Kích hoạt điện trở kéo lên nội vi cho chân nút nhấn để không cần gắn điện trở ngoài

__pinMode(pinSW, INPUT_PULLUP);

}

void loop() {

__// Đọc giá trị Analog của trục X và trục Y (giá trị từ 0 đến 1023)

__int xVal = analogRead(pinX);

__int yVal = analogRead(pinY);

__// Đọc trạng thái nút nhấn (Trả về LOW khi bị ấn xuống)

__int btnState = digitalRead(pinSW);

__// In tọa độ và trạng thái nút bấm lên màn hình máy tính

__Serial.print("Trục X: "); Serial.print(xVal);

__Serial.print(" | Trục Y: "); Serial.print(yVal);

__Serial.print(" | Nút bấm: ");

__if (btnState == LOW) {

____Serial.println("ĐANG ẤN");

__} else {

____Serial.println("Thả tự do");

__}

__delay(200); // Tránh tràn dữ liệu, cập nhật 5 lần mỗi giây

}
