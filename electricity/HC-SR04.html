Thông tin sơ bộ về HC-SR04:

Thông tin cơ bản:

\(\bullet\) Đây là 1 con đo khoảng cách 1 vật thể bằng cách đo thời gian sóng âm dội về

\(\bullet\) Điện áp hoạt động: 5V

\(\bullet\) Không cần gắn thêm điện trở

\(\bullet\) Sơ đồ nối chân cơ bản(giữa chân cảm biến-arduino)

- chân Trig nối với chân 3(hoặc bất kì chân digital nào của arduino)

- chân Echo nối với chân 4(hoặc bất kì chân digital nào của arduino)

- Cực dương(VCC) nối với đầu 5V

- Cực âm(GND) nối với đầu GND

- Code mẫu:

// Định nghĩa các chân kết nối cảm biến HC-SR04 trên Arduino

const int trigPin = 3; // Chân phát sóng (Trig) nối với chân Digital 3

const int echoPin = 4; // Chân nhận sóng (Echo) nối với chân Digital 4

// Khai báo các biến lưu trữ thời gian và khoảng cách

long duration;

int distance;

void setup() {

__pinMode(trigPin, OUTPUT); // Chân Trig cấu hình là Đầu Ra

__pinMode(echoPin, INPUT); // Chân Echo cấu hình là Đầu Vào

__Serial.begin(9600); // Mở cổng Serial Monitor xem kết quả

}

void loop() {

__// Xóa sạch chân trigPin trước khi phát xung

__digitalWrite(trigPin, LOW);

__delayMicroseconds(2);

__// Kích hoạt chân trigPin phát xung siêu âm trong 10 micro giây

__digitalWrite(trigPin, HIGH);

__delayMicroseconds(10);

__digitalWrite(trigPin, LOW);

__// Đo thời gian (micro giây) sóng siêu âm phản hồi lại chân echoPin

__duration = pulseIn(echoPin, HIGH);

__// Tính toán khoảng cách đến vật cản (đơn vị: cm)

__distance = duration * 0.0343 / 2;

__// In kết quả đo được lên Serial Monitor trên máy tính

__Serial.print("Khoang cach: ");

__Serial.print(distance);

__Serial.println(" cm");

__delay(500); // Đợi 0.5 giây trước khi đo lượt tiếp theo

__// Đo khá ổn định tầm 40-50cm

}
