Thông tin sơ bộ về HW-080:

Thông tin cơ bản:

\(\bullet\) Đây là 1 con cảm biến đo độ ẩm trong đất

\(\bullet\) Điện áp hoạt động: 5V

\(\bullet\) Không cần gắn thêm điện trở

\(\bullet\) Sơ đồ nối chân cơ bản(giữa chân cảm biến-arduino)

- Chân D0 nối với chân 7(hoặc bất kì chân digital nào của arduino)

- Chân A0 nối với chân A0(hoặc bất kì chân analog nào của arduino)

- Cực dương(VCC) nối với đầu 5V

- Cực âm(GND) nối với đầu GND

- Code mẫu:

// Định nghĩa chân kết nối cảm biến mưa HW-080

const int analogPin = A0; // Chân A0 nối vào Analog A0 trên Arduino

const int digitalPin = 7; // Chân D0 nối vào Digital 7 trên Arduino

int rainAnalog = 0; // Biến lưu giá trị lượng nước thô (0 - 1023)

int rainDigital = 0; // Biến lưu trạng thái logic (0 hoặc 1)

void setup() {

__pinMode(digitalPin, INPUT); // Cấu hình chân D0 là Đầu Vào

__Serial.begin(9600); // Mở cổng Serial Monitor xem kết quả

}

void loop() {

__// Đọc dữ liệu từ cảm biến

__rainAnalog = analogRead(analogPin);

__rainDigital = digitalRead(digitalPin);

__// In giá trị Analog (Càng nhiều nước, giá trị càng thấp)

__Serial.print("Gia tri am uot: ");

__Serial.print(rainAnalog);

__Serial.print(" | ");

__// Kiểm tra cảnh báo mưa từ chân D0

__if (rainDigital == LOW) { // Khi dính nước, mạch kéo chân D0 xuống LOW

____Serial.println("Trang thai: PHA'T HIE^.N CO' MU'A / NUO*'C!");

__} else {

____Serial.println("Trang thai: Kho rao");

__}

__delay(500); // Đọc dữ liệu sau mỗi 0.5 giây

}
