Thông tin sơ bộ về Water sensor:

Thông tin cơ bản:

\(\bullet\) Đây là 1 cảm biến đo mực nước

\(\bullet\) Điện áp hoạt động: 5V

\(\bullet\) Không cần gắn thêm điện trở

\(\bullet\) Sơ đồ nối chân cơ bản(giữa chân cảm biến-arduino)

- Chân Signal(S) nối với chân A0

- Cực dương(+) nối với đầu 5V

- Cực âm(-) nối với đầu GND

- Code mẫu:

// Định nghĩa chân kết nối cảm biến mực nước

const int waterPin = A0; // Chân Signal nối với Analog A0 trên Arduino

int waterValue = 0; // Biến lưu giá trị đo được

void setup() {

__Serial.begin(9600); // Mở cổng Serial Monitor với tốc độ 9600 baud

}

void loop() {

__// Đọc giá trị Analog từ cảm biến (từ 0 đến 1023)

__waterValue = analogRead(waterPin);

__// In giá trị thô ra Serial Monitor để theo dõi

__Serial.print("Muc nuoc do duoc: ");

__Serial.println(waterValue);

__// Phân loại mức nước dựa trên giá trị đọc được

__if (waterValue < 10) {

____Serial.println("--> Trạng thái: Cảm biến đang khô ráo");

__} else if (waterValue <= 350) {

____Serial.println("--> Trạng thái: Mức nước THẤP");

__} else if (waterValue <= 600) {

____Serial.println("--> Trạng thái: Mức nước TRUNG BÌNH");

__} else {

____Serial.println("--> Trạng thái: Mức nước CAO / NGẬP ĐẦY");

__}

__delay(1000); // Đọc lại sau mỗi 1 giây

}
