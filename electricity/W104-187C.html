Thông tin sơ bộ về W104-187C:

Thông tin cơ bản:

\(\bullet\) Đây là 1 con cảm biến phát hiện tiếng động và đo cường độ âm thanh

\(\bullet\) Điện áp hoạt động: 5V

\(\bullet\) Không cần gắn thêm điện trở

\(\bullet\) Sơ đồ nối chân cơ bản(giữa chân cảm biến-arduino)

- Chân A0 nối với chân A0(hoặc bất kì chân Analog nào của arduino)

- Chân D0 nối với chân D7(hoặc bất kì chân Digital nào của arduino)

- Cực dương(+) nối với đầu 5V

- Cực âm(G) nối với đầu GND

- Code mẫu:

// Khai báo chân kết nối cảm biến âm thanh 4 chân

const int analogPin = A0; // Chân A0 của cảm biến nối với Analog A0

const int digitalPin = 7; // Chân D0 của cảm biến nối với Digital 7

int soundValue = 0; // Biến lưu giá trị cường độ âm thanh thô

int soundStatus = 0; // Biến lưu trạng thái bật/tắt (0 hoặc 1)

void setup() {

__pinMode(digitalPin, INPUT); // Cấu hình chân D0 là Đầu Vào

__Serial.begin(9600); // Mở Serial Monitor để quan sát dữ liệu

}

void loop() {

__// Đọc dữ liệu từ cả 2 chân A0 và D0 cùng lúc

__soundValue = analogRead(analogPin);

__soundStatus = digitalRead(digitalPin);

__// In giá trị cường độ âm thanh liên tục lên màn hình

__Serial.print("Cuong do am thanh: ");

__Serial.print(soundValue);

__Serial.print(" | ");

__// Kiểm tra trạng thái kích hoạt từ chân D0

__if (soundStatus == LOW) { // Khi âm thanh vượt ngưỡng, module kéo D0 xuống LOW

____Serial.println("Trang thai: VUOT NGUONG AM THANH!");

__} else {

____Serial.println("Trang thai: Yen tinh");

__}

__delay(100); // Đọc dữ liệu mỗi 0.1 giây để bắt kịp sóng âm

}
