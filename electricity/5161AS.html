\(\bullet\) Đây là 1 con đo khoảng cách 1 vật thể bằng cách đo thời gian sóng âm dội về

\(\bullet\) Sơ đồ nối chân cơ bản(giữa chân cảm biến-arduino)

- chân a nối với điện trở 220\(\Omega\) nối với chân 2(hoặc bất kì chân digital nào của arduino)

- chân b nối với điện trở 220\(\Omega\) nối với chân 3(hoặc bất kì chân digital nào của arduino)

- chân c nối với điện trở 220\(\Omega\) nối với chân 4(hoặc bất kì chân digital nào của arduino)

- chân d nối với điện trở 220\(\Omega\) nối với chân 5(hoặc bất kì chân digital nào của arduino)

- chân e nối với điện trở 220\(\Omega\) nối với chân 6(hoặc bất kì chân digital nào của arduino)

- chân f nối với điện trở 220\(\Omega\) nối với chân 7(hoặc bất kì chân digital nào của arduino)

- chân g nối với điện trở 220\(\Omega\) nối với chân 8(hoặc bất kì chân digital nào của arduino)

- Cực âm(cathode) nối với đầu GND

- Code mẫu:

// Khai báo các chân Arduino nối với các đoạn LED A, B, C, D, E, F, G

const int segmentPins[] = {2, 3, 4, 5, 6, 7, 8};

__

// Mảng mã hóa các số từ 0-9 cho LED Âm Chung (1 là SÁNG, 0 là TẮT)

const byte numPatterns[10] = {

__0b00111111, // Số 0 (Bật A,B,C,D,E,F)

__0b00000110, // Số 1 (Bật B,C)

__0b01011011, // Số 2 (Bật A,B,D,E,G)

__0b01001111, // Số 3 (Bật A,B,C,D,G)

__0b01100110, // Số 4 (Bật B,C,F,G)

__0b01101101, // Số 5 (Bật A,C,D,F,G)

__0b01111101, // Số 6 (Bật A,C,D,E,F,G)

__0b00000111, // Số 7 (Bật A,B,C)

__0b01111111, // Số 8 (Bật tất cả các nét)

__0b01101111 // Số 9 (Bật A,B,C,D,F,G)

};

__

void setup() {

__// Cấu hình tất cả các chân LED là OUTPUT

__for (int i = 0; i < 7; i++) {

____pinMode(segmentPins[i], OUTPUT);

__}

}

__

void loop() {

__// Chạy vòng lặp hiển thị từ số 0 đến số 9

__for (int num = 0; num < 10; num++) {

____displayNumber(num);

____delay(1000); // Chờ 1 giây rồi đổi số

__}

}

__

// Hàm xuất tín hiệu ra các chân LED

void displayNumber(int num) {

__byte pattern = numPatterns[num];

__for (int i = 0; i < 7; i++) {

____// Trích xuất từng bit trong mã để bật/tắt chân tương ứng

____digitalWrite(segmentPins[i], bitRead(pattern, i));

__}

}
