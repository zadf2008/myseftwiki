\(\bullet\) switch là cấu trúc rẽ nhánh

\(\bullet\) các kiểu giá trị nhận: các kiểu dữ liệu số nguyên, bool và char

\(\bullet\) Ưu điểm: rất nhanh khi có nhiều kết quả

\(\bullet\) Nhược điểm: kém linh hoạt

\(\bullet\) Cấu trúc:

switch ([giá trị đưa vô để kiểm tra]) {

__case [giá trị 1]:

__//khối lệnh

__break;

__case [giá trị 2]:

__//khối lệnh

__break;

__...

__default:

__//khối lệnh

__break;

}

Ví dụ:

#include <iostream>

using namespace std;

____

int main() {

__int a;

__cin >> a;

__switch (a) {

____case 1:

______cout << "ban nhap so 1"

______break;

____case 2:

______cout << "ban nhap so 2"

______break;

____default:

______cout << "so ban nhap chua duoc cap nhat"

____break;

__}

}
