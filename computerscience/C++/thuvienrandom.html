\(\bullet\) Thư viện tạo ra 1 số ngẫu nhiên

Ví dụ 1 (tạo ra 1 số nguyên):

#include <iostream>

#include <random>

____

int main() {

__std::random_device rd;__//lấy một hạt giống từ phần cứng

____

__std::mt19937 gen(rd());__//tạo một engine từ hạt giống

____

__std::uniform_int_distribution<> dis1(1, 100);__//ngẫu nhiên ra số nguyên

____

__int so_ngau_nhien = dis1(gen);

____

__std::cout << "So ngau nhien cua ban la: " << so_ngau_nhien << std::endl;

__return 0;

}

Ví dụ 2(tạo ra 1 số thực):

#include <iostream>

#include <random>

____

int main() {

__std::random_device rd;__//lấy một hạt giống từ phần cứng

____

__std::mt19937 gen(rd());__//tạo engine sinh số ngẫu nhiên

____

__std::uniform_real_distribution<> dis2(1, 100);__//thiết lập khoảng số thực từ 1 đến dưới 100

____

__double so_thuc_ngau_nhien = dis2(gen);__//lấy một số thực ngẫu nhiên

____

__std::cout << "So thuc ngau nhien cua ban la: " << so_thuc_ngau_nhien << std::endl;

__return 0;

}

Ví dụ 3(ngẫu nhiên đúng sai):

#include <iostream>

#include <random>

____

int main() {

__std::random_device rd;__//lấy một hạt giống từ phần cứng

____

__std::mt19937 gen(rd());__//tạo engine sinh số ngẫu nhiên

____

__std::bernoulli_distribution dis3(0.3);__//30% true, 70% false

____

__bool thanh_cong = dis3(gen);__//lấy kết quả đúng hoặc sai

____

__std::cout << "Co thanh cong khong: " << thanh_cong << std::endl;

__return 0;

}
