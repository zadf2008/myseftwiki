\(\bullet\) Vẽ một hình bằng hàm:

import numpy as np

import matplotlib.pyplot as plt

# 1. Tạo dữ liệu cho góc quay Theta (từ 0 đến 2*pi)

----theta = np.linspace(0, 2 * np.pi, 500)

# 2. Tạo dữ liệu cho bán kính r (Công thức tạo hình bông hoa 3 cánh)

----r = np.sin(3 * theta)

# 3. Khởi tạo biểu đồ với cấu hình cực (subplot_kw={'projection': 'polar'})

----fig, ax = plt.subplots(subplot_kw={"projection": "polar"}, figsize=(6, 6))

# 4. Vẽ đường cong lên biểu đồ cực

----ax.plot(theta, r, color="purple", linewidth=2.5, label="Bông hoa 3 cánh")

----

# 5. Tô màu bên trong cánh hoa cho đẹp mắt (tùy chọn)

----ax.fill(theta, r, color="pink", alpha=0.5)

----

# 6. Tinh chỉnh hiển thị (tùy chọn)

----ax.set_rmax(1.2) # Giới hạn bán kính lớn nhất hiển thị trên vòng tròn

----ax.set_rticks([0.5, 1.0]) # Vẽ các đường tròn đồng tâm tại r = 0.5 và r = 1.0

----ax.set_rlabel_position(-22.5) # Góc để hiển thị nhãn của bán kính

----ax.grid(True) # Hiển thị lưới tọa độ tròn

plt.title("Ví dụ về Biểu đồ Tọa độ Cực (Polar Plot)", va="bottom", fontsize=14)

plt.show()

\(\bullet\) Đồ thị tuần suất

import matplotlib.pyplot as plt

----

# 1. Tạo dữ liệu giả lập: Mốc thời gian xảy ra lỗi trên 3 Server

# Mỗi danh sách con đại diện cho các mốc thời gian lỗi của một Server

server_A = [12, 15, 34, 55, 78, 90] # Server A lỗi vào giây thứ 12, 15...

server_B = [5, 22, 23, 24, 60, 85, 95] # Server B lỗi dồn dập ở giây 22-24

server_C = [40, 42, 45, 50, 88] # Server C lỗi chủ yếu ở giữa chu kỳ

----

tat_ca_su_kien = [server_A, server_B, server_C]

----

# 2. Khởi tạo khung vẽ

fig, ax = plt.subplots(figsize=(10, 4))

----

# 3. Vẽ biểu đồ sự kiện bằng hàm plt.eventplot()

# orientation="horizontal" nghĩa là các vạch sự kiện xếp theo hàng ngang

ax.eventplot(tat_ca_su_kien, orientation="horizontal", colors=["red", "blue", "green"], lineoffsets=1, linelengths=0.6)

----

# 4. Trang trí biểu đồ cho dễ đọc

ax.set_title("Nhật Ký Sự Kiện Lỗi Trên Các Máy Chủ", fontsize=14, fontweight="bold")

ax.set_xlabel("Thời gian (Giây)", fontsize=12)

ax.set_ylabel("Danh sách Máy chủ", fontsize=12)

----

# Thay thế các số 0, 1, 2 mặc định trên trục Y thành tên Server cụ thể

ax.set_yticks([0, 1, 2])

ax.set_yticklabels(["Server A", "Server B", "Server C"])

----

ax.grid(True, axis="x", linestyle="--", alpha=0.5) # Bật lưới dọc để dễ gióng thời gian

----

plt.tight_layout()

----

plt.show()
