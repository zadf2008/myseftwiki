Hàm __str__:

class Player:

----def __init__(self, name, level):

--------self.name = name

--------self.level = level

----def __str__(self):

--------return f"Player: {self.name} | Level: {self.level}"

----

char1 = Player("zadf2008", 20)

print(char1)

#__str__ có tác dụng thay đổi nội dung khi print(object)

Hàm __len__:

class Bag:

----def __init__(self, items):

--------self.items = items

----def __len__(self):

--------return len(self.items)

----

bag = Bag(["A", "B", "C"])

print(len(bag))

#__len__ có tác dụng thay đổi kết quả của len(object)

Hàm __truediv__ và __floordiv__ và __mod__:

class Number:

----def __init__(self, value):

--------self.value = value

----def __truediv__(self, other):

--------return Number(self.value / other.value)

----def __floordiv__(self, other):

--------return Number(self.value // other.value)

----def __mod__(self, other):

--------return Number(self.value % other.value)

----

a = Number(10)

b = Number(3)

print((a / b).value)-----#Phép chia bình thường

print((a // b).value)----#Phép chia rồi làm tròn xuống

print((a % b).value)-----#Phép chia rồi lấy phần dư

Hai phép nhận đặc biệt

class MayTinh:

----def __call__(self, *args, **kwargs):

--------if args:# nếu không trống thì thỏa

------------tong = sum(args) #tính tổng các số

------------print(f"Tổng các số: {tong}")

--------if kwargs:

------------print("Thông tin:")

------------for ten, gia_tri in kwargs.items():

----------------print(f"{ten}: {gia_tri}")

# *args có tác dụng nhận nhiều phần tử từ số, phân số và chuỗi, hoặc cả danh sách

# **kwargs có tác dụng nhận các biến

----

may_tinh = MayTinh()

may_tinh(10, 20, 30, 40)

may_tinh(tuoi="Robot", tuoi=2, trang_thai="Hoạt động")

may_tinh(10, 20, ten="Robot", tuoi=2)
