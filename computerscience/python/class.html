Ví dụ kiểu 1:

class Call():

----def __init__(self, name, age):

--------self.name = name

--------self.age = age

--------print(f"Hello {self.name}")

--------print(f"Your age is {self.age}")

----

Call(input("What's your name? My name is "), int(input("How old are you? My age is ")))
